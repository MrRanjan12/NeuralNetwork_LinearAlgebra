# 2. Sigmoid Activation Function (for introducing non-linearity)
def sigmoid(x):
    return 1 / (1 + np.exp(-x))

# 3. Network Initialization (Random Weights and Biases with seed 42)
np.random.seed(42)

# 4. Forward Propagation (The core math: Matrix Multiplication)
def forward_propagation(X, weights, biases):
    z1 = np.dot(X, weights['w1']) + biases['b1']
    a1 = sigmoid(z1)
    z2 = np.dot(a1, weights['w2']) + biases['b2']
    return z2
